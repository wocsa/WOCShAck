Python — List Your CSS Projects

import requests

API_KEY = "vrc_your_api_key_here"
BASE_URL = "http://localhost:8000/api"

headers = {"X-API-Key": API_KEY}

response = requests.get(f"{BASE_URL}/css/my/", headers=headers)
projects = response.json()

for project in projects["data"]:
    print(f"{project['name']} - {project['category']}")

JavaScript — Fetch Analytics

const API_KEY = 'vrc_your_api_key_here';

async function getAnalytics(days = 30) {
    const response = await fetch(
        `/api/analytics/?days=${days}`,
        { headers: { 'X-API-Key': API_KEY } }
    );
    const data = await response.json();
    console.log(`Revenue: ${data.data.total_revenue} NE`);
    console.log(`Sales: ${data.data.total_sales}`);
    return data;
}

Python — Webhook Handler (Flask)

import hmac
import hashlib
from flask import Flask, request, jsonify

app = Flask(__name__)
WEBHOOK_SECRET = "your_webhook_secret"

@app.route("/webhook", methods=["POST"])
def handle_webhook():
    signature = request.headers.get("X-Webhook-Signature", "")
    payload = request.get_data()

    expected = hmac.new(
        WEBHOOK_SECRET.encode(),
        payload,
        hashlib.sha256
    ).hexdigest()

    if not hmac.compare_digest(expected, signature):
        return jsonify({"error": "Invalid signature"}), 401

    event = request.json
    print(f"Received event: {event['event']}")

    if event["event"] == "sale.completed":
        print(f"New sale! Amount: {event['data']['amount']} NE")

    return jsonify({"status": "ok"}), 200
